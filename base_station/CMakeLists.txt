# ==============================================================================
# BASE STATION - BUILD CONFIGURATION (CMake)
# ==============================================================================
#
# WHAT IS THIS FILE?
# This project is built using CMake, a tool that reads a file like this one
# (called "CMakeLists.txt") and turns it into the actual low-level build
# instructions (a Makefile, or in this project's case, a "Ninja" build
# file) that compile every .c file into machine code and link them all
# together into the final program. In other words: this file is the
# recipe that says "these are the source files, these are the libraries
# we need, here is how to put it all together" for the base station
# program.
#
# Generated Cmake Pico project file

cmake_minimum_required(VERSION 3.13)

# Which version of the C and C++ language standard to compile against.
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
# Also write out a "compile_commands.json" file describing exactly how each
# file was compiled - many code editors (like VS Code) use this to give
# accurate autocomplete and error-checking.
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Initialise pico_sdk from installed location
# (note this can come from environment, CMake cache etc)

# == DO NOT EDIT THE FOLLOWING LINES for the Raspberry Pi Pico VS Code Extension to work ==
# This block locates the Raspberry Pi Pico SDK (the official library of
# code for talking to the Pico's hardware - GPIO pins, USB, etc.) that was
# installed on this computer, and pulls in some VS-Code-extension-specific
# setup if it is present. It is boilerplate generated by the Pico project
# tooling, not something specific to this project's own logic.
if(WIN32)
    set(USERHOME $ENV{USERPROFILE})
else()
    set(USERHOME $ENV{HOME})
endif()
set(sdkVersion 2.3.0)
set(toolchainVersion 15_2_Rel1)
set(picotoolVersion 2.3.0)
set(picoVscode ${USERHOME}/.pico-sdk/cmake/pico-vscode.cmake)
if (EXISTS ${picoVscode})
    include(${picoVscode})
endif()
# ====================================================================================

# Which physical board we are building for. "pico2_w" is the Raspberry Pi
# Pico 2 W - the "W" model that has the extra CYW43 chip needed for Wi-Fi,
# which this program needs in order to run its web server.
set(PICO_BOARD pico2_w CACHE STRING "Board type")

# Wifi credentials
#
# These two settings are deliberately left BLANK here rather than
# hard-coded, and "CACHE STRING" means "this is a setting the person
# running cmake can override" - for example with:
#     cmake .. -DWIFI_SSID="my-network" -DWIFI_PASSWORD="my-password"
# This keeps the real Wi-Fi password out of the source code / git history.
# See the target_compile_definitions() call further down for how these
# values actually make their way into the compiled program (as WIFI_SSID
# and WIFI_PASSWORD, used in network/wifi_station.c).
set(WIFI_SSID "" CACHE STRING "Wi-Fi network name")
set(WIFI_PASSWORD "" CACHE STRING "Wi-Fi password")

# Set the path for the FreeRTOS kernel
#
# FreeRTOS (the "operating system" that lets this program run several
# things at once - see app/main.c for a fuller explanation) is not part of
# the Pico SDK itself; it lives in its own folder, checked out separately
# alongside this project (in ../external/FreeRTOS-Kernel). This just
# records where to find it.
set(FREERTOS_KERNEL_PATH
    ${CMAKE_CURRENT_LIST_DIR}/../external/FreeRTOS-Kernel
)

# Set the path for the NRF24L01 driver
#
# Similarly, the low-level code that knows how to talk to the nRF24L01+
# radio chip over SPI is a separate, third-party ("vendored") library,
# living in ../external/nrf24l01. This project's own code
# (common/nrf24l01/driver_nrf24l01_temperature.c) is a thin layer built on
# top of it with the specific settings this project needs.
set(NRF24L01_PATH
    ${CMAKE_CURRENT_LIST_DIR}/../external/nrf24l01
)

# Set the path for code shared between sensor and base_station
#
# This project actually has TWO separate programs: this one (base_station)
# and a second one (sensor, built from its own, separate CMakeLists.txt),
# running on two different physical boards. Some code is genuinely useful
# to both - for example, FreeRTOS crash-handling hooks, the nRF24L01 radio
# driver wrapper, and small text-formatting helpers - so that code lives
# in one shared folder, ../common, instead of being copied into both
# programs separately.
set(COMMON_PATH
    ${CMAKE_CURRENT_LIST_DIR}/../common
)

# Pull in Raspberry Pi Pico SDK (must be before project)
include(pico_sdk_import.cmake)

# Pull in the FreeRTOS kernel
#
# This particular path (RP2350_ARM_NTZ) is the specific FreeRTOS "port" -
# the small piece of FreeRTOS written specifically for this chip's
# processor architecture (the RP2350, as used in the Pico 2) - needed so
# FreeRTOS knows how to actually switch between tasks on THIS hardware.
include(
    "${FREERTOS_KERNEL_PATH}/portable/ThirdParty/Community-Supported-Ports/GCC/RP2350_ARM_NTZ/FreeRTOS_Kernel_import.cmake"
)

# This declares the actual CMake "project": its name (base_station) and
# which programming languages it uses (C, C++, and ASM for a few very
# low-level startup routines).
project(base_station C CXX ASM)

# Initialise the Raspberry Pi Pico SDK
pico_sdk_init()

# Add executable. Default name is the project name, version 0.1
#
# This is the heart of the file: the full list of every .c/.cpp source
# file that gets compiled and linked together into the final
# "base_station" program. Files without a "${...}/" prefix live directly
# in this project's own folders (app/, temperature/, network/, web/);
# files prefixed with ${COMMON_PATH} or ${NRF24L01_PATH} live in the
# shared folders explained above.
add_executable(base_station
    app/main.c
    ${COMMON_PATH}/task_utils.c
    ${COMMON_PATH}/temperature_format.c
    ${COMMON_PATH}/hooks.cpp
    ${COMMON_PATH}/nrf24l01/driver_nrf24l01_interface.c
    ${COMMON_PATH}/nrf24l01/driver_nrf24l01_temperature.c
    ${NRF24L01_PATH}/src/driver_nrf24l01.c
    temperature/temperature_receiver.c
    temperature/temperature_store.c
    network/base_station_network.c
    network/wifi_station.c
    web/web_server.c
    web/temperature_ssi.c
)

# The name and version number embedded into the compiled program, visible
# for example via USB device information.
pico_set_program_name(base_station "base_station")
pico_set_program_version(base_station "0.1")

# Modify the below lines to enable/disable output over UART/USB
#
# These turn on printf()/debug output over BOTH the Pico's UART pins
# (physical serial wires) and its USB port, so a person can plug in and
# watch the debug log either way.
pico_enable_stdio_uart(base_station 1)
pico_enable_stdio_usb(base_station 1)

# This creates a small, separate helper "library" that will hold the web
# page content (see pico_set_lwip_httpd_content() right below) as raw data
# baked into the program.
pico_add_library(base_station_http_content NOFLAG)

# pico_set_lwip_httpd_content() is a Pico SDK helper that takes real files
# from disk (here, everything in network/http/) and, at build time,
# converts them into a big C array of raw bytes that gets compiled
# directly into the program - see lwipopts.h's HTTPD_FSDATA_FILE comment
# for why: this tiny microcontroller has no real filesystem to serve files
# from at runtime, so every file the web server can serve has to already
# be "baked in" ahead of time, at build time, instead.
pico_set_lwip_httpd_content(
    base_station_http_content
    INTERFACE
    ${CMAKE_CURRENT_LIST_DIR}/network/http/404.html
    ${CMAKE_CURRENT_LIST_DIR}/network/http/index.shtml
    ${CMAKE_CURRENT_LIST_DIR}/network/http/temphist.json
    ${CMAKE_CURRENT_LIST_DIR}/network/http/ack.json
)


# This is how the WIFI_SSID / WIFI_PASSWORD cmake settings from near the
# top of this file actually reach the C code: it defines them as C
# preprocessor macros (as if "#define WIFI_SSID "my-network"" had been
# written directly into every source file), which network/wifi_station.c
# then simply uses as if they were ordinary string constants.
target_compile_definitions(base_station PRIVATE
    WIFI_SSID="${WIFI_SSID}"
    WIFI_PASSWORD="${WIFI_PASSWORD}"
)

# Add the standard include files to the build
#
# This is the list of folders the compiler should search when a source
# file writes something like #include "temperature_store.h" - without an
# entry here for the folder a header actually lives in, the compiler would
# not be able to find it. Notice this project's own subfolders
# (network/, temperature/, web/) are listed as flat folders - this is why,
# for example, temperature_receiver.c can simply write
# #include "temperature_store.h" instead of needing a longer relative path.
target_include_directories(base_station PRIVATE
        ${CMAKE_CURRENT_LIST_DIR}
        ${CMAKE_CURRENT_LIST_DIR}/network
        ${CMAKE_CURRENT_LIST_DIR}/temperature
        ${CMAKE_CURRENT_LIST_DIR}/web
        ${COMMON_PATH}
        ${COMMON_PATH}/nrf24l01
        ${NRF24L01_PATH}/src
        ${NRF24L01_PATH}/interface
        ${PICO_LWIP_CONTRIB_PATH}/apps/httpd
)

# Add the standard library to the build
#
# The pre-built libraries this program is linked against:
#   pico_stdlib              - the core Pico SDK (GPIO, timing, stdio, etc.)
#   hardware_spi              - talking over SPI, used by the radio driver
#   FreeRTOS-Kernel /
#   FreeRTOS-Kernel-Heap4      - FreeRTOS itself, plus its "Heap4" memory
#                                allocator (one of several memory-management
#                                strategies FreeRTOS offers; Heap4 is a
#                                general-purpose one suitable for this
#                                project's needs)
#   pico_cyw43_arch_lwip_sys_freertos
#                              - the Pico SDK's integration code that makes
#                                the CYW43 Wi-Fi chip and lwIP networking
#                                library run properly as FreeRTOS tasks
#   pico_lwip_http             - lwIP's own small built-in web server
#                                (the "httpd" mentioned throughout web/)
#   base_station_http_content  - the web page content library created by
#                                pico_set_lwip_httpd_content() above
target_link_libraries(base_station
    pico_stdlib
    hardware_spi
    FreeRTOS-Kernel
    FreeRTOS-Kernel-Heap4
    pico_cyw43_arch_lwip_sys_freertos
    pico_lwip_http
    base_station_http_content
)


# Generates the additional output files alongside the main program: a
# .uf2 file (the format used to flash the Pico by drag-and-drop), a .hex
# file, a memory map, and so on.
pico_add_extra_outputs(base_station)
